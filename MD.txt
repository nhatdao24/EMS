    ADDRESS NVARCHAR(500) ,   -- Địa chỉ
    AREA NVARCHAR(100) ,      -- Khu vực/Vùng

    CREATE_BY varchar(50),
    UPDATE_BY  varchar(50),
    CREATE_DATE  datetime,
    UPDATE_DATE  datetime,

    DELETE_DATE datetime,
    DELETE_BY varchar(50)
);
